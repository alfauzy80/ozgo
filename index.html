      if (btn.dataset.openAuth) { state.portal = 'auth'; state.authTab = btn.dataset.openAuth; render(); return; }

      if (btn.dataset.dfilter) { state.adminDriverFilter = btn.dataset.dfilter; render(); return; }

        state.orderModalService = null; state.ozbudemenOpen = false;
        state.activeChatOrderId = null; state.activeDAChatDriverId = null;
        state.editingCustomerUid = null; state.editingDriverUid = null;
        renderModal(); return;
      }

        state.activeChatOrderId = btn.dataset.openChat;
        state.activeChatRole = btn.dataset.chatRole || 'customer';
        state.unread[state.activeChatOrderId] = 0;
        render(); return;

        state.activeDAChatDriverId = btn.dataset.openDachat;
        state.activeDAChatRole = btn.dataset.darole || 'driver';
        state.unreadDA[state.activeDAChatDriverId] = 0;
        render(); return;

        state.soundEnabled = !state.soundEnabled;

        state.isSuperAdmin = false;

          d.approvalStatus = btn.dataset.status;

      }

          showToast('Sidik jari asli terverifikasi! Selamat datang Super Admin OZgo.', 'success'); render();

          showToast(state.adminBioErr, 'error'); render();
        }
        return;

        state.isSuperAdmin = true; state.activeDriverUid = CONFIG.superAdminDriverUid; state.portal = 'admin'; state.adminPwdVerified = false; state.adminBioErr = '';

      }

        state.adminPwdVerified = false; state.adminBioErr = ''; state.adminBioIframeBlocked = false; render(); return;
      }
      if (btn.id === 'btn-admin-bio-enroll') {
        const res = await enrollRealFingerprint();
        showToast(res.ok ? 'Sidik jari asli perangkat berhasil didaftarkan!' : (res.error || 'Gagal mendaftarkan sidik jari.'), res.ok ? 'success' : 'error');
        render(); return;
      }
      if (btn.id === 'btn-admin-bio-reset') {
        localStorage.removeItem(WEBAUTHN_KEY); state.adminBioCred = null;
        showToast('Sidik jari perangkat ini dihapus.', 'info'); render(); return;
      }
      if (btn.id === 'btn-google-driver' || btn.id === 'btn-google-admin') {
        if (!fbAuth || !googleProvider) { showToast('Gunakan mode demo jika domain belum terdaftar di Firebase Auth', 'info'); return; }
        try {
          const res = await signInWithPopup(fbAuth, googleProvider);
          const isSA = (res.user.email || '').toLowerCase() === CONFIG.superAdminEmail.toLowerCase();
          state.isSuperAdmin = isSA;
          state.activeDriverUid = isSA ? CONFIG.superAdminDriverUid : res.user.uid;
          state.portal = btn.id === 'btn-google-admin' && isSA ? 'admin' : 'driver';
          render();
        } catch { showToast('Login Google dibatalkan atau domain belum diotorisasi', 'error'); }
        return;
      }
      if (btn.id === 'btn-logout-portal') {
        stopGps();
        if (fbAuth) { try { await signOut(fbAuth); } catch {} }
        state.isSuperAdmin = false; state.activeDriverUid = null; state.portal = 'customer';
        showToast('Berhasil keluar', 'info'); render(); return;
      }
      if (btn.id === 'btn-driver-online') {
        const drv = getActiveDriver();
        if (drv && drv.approvalStatus === 'approved') { drv.isOnline = true; persistAll(); showToast('Status Anda sekarang ONLINE', 'success'); render(); }
        return;
      }
      if (btn.id === 'btn-driver-offline') {
        const drv = getActiveDriver();
        if (drv) { drv.isOnline = false; stopGps(); persistAll(); showToast('Status Anda sekarang OFFLINE', 'info'); render(); }
        return;
      }
      if (btn.id === 'btn-gps-sim') { toggleGpsSim(); return; }
      if (btn.id === 'btn-copy-html-bottom') {
        const src = '<!doctype html>\n' + document.documentElement.outerHTML;
        try { await navigator.clipboard.writeText(src); } catch {}
        state.htmlCopied = true;
        showToast('Kode HTML berhasil disalin!', 'success');
        render();
        setTimeout(() => { state.htmlCopied = false; render(); }, 2500);
        return;
      }
      if (btn.id === 'btn-download-html-bottom') {
        const src = '<!doctype html>\n' + document.documentElement.outerHTML;
        const blob = new Blob([src], { type: 'text/html;charset=utf-8' });
        const a = document.createElement('a');
        a.href = URL.createObjectURL(blob); a.download = 'ozgo.html';
        document.body.appendChild(a); a.click(); a.remove();
        return;
      }

      if (e.target.id === 'svc-search') { state.searchQuery = e.target.value; render(); document.getElementById('svc-search')?.focus(); }
      if (e.target.id === 'admin-driver-search') { state.adminSearch = e.target.value; render(); document.getElementById('admin-driver-search')?.focus(); }
      if (e.target.id === 'admin-user-search') { state.adminUserSearch = e.target.value; render(); document.getElementById('admin-user-search')?.focus(); }
    });

    document.addEventListener('change', (e) => {
      if (e.target.dataset.assignOrder) {
        const o = state.orders.find((x) => x.orderId === e.target.dataset.assignOrder);
        const d = state.drivers.find((x) => x.uid === e.target.value);
        if (o && d) {
          o.assignedDriverId = d.uid; o.assignedDriverName = d.name; o.status = 'assigned';
          persistAll(); showToast(`Pesanan #${o.orderId} ditugaskan ke ${d.name}`, 'success'); render();
        }
      }
    });

    document.addEventListener('submit', async (e) => {
      e.preventDefault();
      if (e.target.id === 'admin-pwd-login-form') {
        const email = document.getElementById('adm-email').value.trim();
        const pass = document.getElementById('adm-pass').value;
        const ok = await verifyAdminCreds(email, pass);
        if (!ok) { state.adminBioErr = 'Email atau password Admin utama OZgo salah.'; showToast(state.adminBioErr, 'error'); render(); return; }
        state.adminBioErr = '';
        if (state.adminBioCred) {
          state.adminPwdVerified = true; render();
          const bio = await loginRealFingerprint();
          if (bio.ok) {
            state.isSuperAdmin = true; state.activeDriverUid = CONFIG.superAdminDriverUid; state.portal = 'admin'; state.adminPwdVerified = false;
            showToast('Sidik jari terverifikasi! Selamat datang Super Admin OZgo.', 'success'); render();
          } else {
            state.adminBioErr = bio.error || 'Verifikasi sidik jari gagal.'; state.adminBioIframeBlocked = Boolean(bio.blockedByIframe); render();
          }
          return;
        }
        if (state.bioSupport.supported) {
          state.adminPwdVerified = true;
          showToast('Email & Password benar. Daftarkan sidik jari asli Anda sekarang.', 'info');
          render(); return;
        }
        state.isSuperAdmin = true; state.activeDriverUid = CONFIG.superAdminDriverUid; state.portal = 'admin';
        showToast('Login Super Admin OZgo berhasil.', 'success'); render();
      } else if (e.target.id === 'user-reconnect-form') {
        const wa = document.getElementById('reconnect-wa').value.trim();
        const norm = normalizeWA(wa);
        const found = state.customers.find((c) => normalizeWA(c.phone) === norm);
        if (!found) { showToast('No. WA belum terdaftar. Silakan daftar user baru.', 'error'); return; }
        state.profile = { uid: found.uid, name: found.name, phone: found.phone, address: found.address };
        persistAll(); showToast(`Selamat datang kembali, ${found.name}! Data Anda langsung terkoneksi.`, 'success'); render();
      } else if (e.target.id === 'user-register-form') {
        const name = document.getElementById('reg-user-name').value.trim();
        const phone = document.getElementById('reg-user-wa').value.trim();
        const address = document.getElementById('reg-user-addr').value.trim();
        if (!name || !isValidWA(phone) || !address) { showToast('Lengkapi Nama Lengkap, No. WA valid, dan Alamat.', 'error'); return; }
        const norm = normalizeWA(phone);
        const existing = state.customers.find((c) => normalizeWA(c.phone) === norm);
        if (existing) {
          state.profile = { uid: existing.uid, name: existing.name, phone: existing.phone, address: existing.address };
          persistAll(); showToast(`No. WA sudah terdaftar. Data ${existing.name} langsung terkoneksi!`, 'success'); render(); return;
        }
        const created = { uid: 'usr_' + Date.now(), name, phone, normalizedPhone: norm, address, createdAtMs: Date.now() };
        state.customers = [created, ...state.customers];
        state.profile = { uid: created.uid, name, phone, address };
        persistAll(); showToast('Pendaftaran user berhasil!', 'success'); render();
      } else if (e.target.id === 'profile-edit-form') {
        const name = document.getElementById('edit-prof-name').value.trim();
        const phone = document.getElementById('edit-prof-phone').value.trim();
        const address = document.getElementById('edit-prof-address').value.trim();
        if (!name || !isValidWA(phone) || !address) { showToast('Data profil belum valid', 'error'); return; }
        const norm = normalizeWA(phone);
        state.profile = { ...state.profile, name, phone, address };
        const idx = state.customers.findIndex((c) => c.uid === state.profile.uid);
        if (idx >= 0) state.customers[idx] = { ...state.customers[idx], name, phone, normalizedPhone: norm, address };
        persistAll(); showToast('Data profil user diperbarui', 'success'); render();
      } else if (e.target.id === 'driver-reg-form') {
        const name = document.getElementById('dreg-name').value.trim();
        const phone = document.getElementById('dreg-phone').value.trim();
        const email = document.getElementById('dreg-email').value.trim() || `${name.toLowerCase().replace(/\s+/g, '.')}@ozgo.id`;
        const vehicleModel = document.getElementById('dreg-model').value.trim();
        const area = document.getElementById('dreg-area').value.trim();
        const newDrv = {
          uid: 'drv_' + Date.now(), name, email, phone, vehicleType: 'motor', vehicleModel, area,
          approvalStatus: 'pending', isOnline: false, lat: CONFIG.center.lat + 0.002, lng: CONFIG.center.lng + 0.002, speed: 0,
        };
        state.drivers = [newDrv, ...state.drivers];
        state.isSuperAdmin = false;
        state.activeDriverUid = newDrv.uid;
        state.portal = 'driver';
        persistAll();
        if (state.soundEnabled) playChatChime();
        showToast('Pendaftaran anda sedang di proses admin OZgo', 'info');
        render();
      } else if (e.target.id === 'order-submit-form') {
        const svc = state.orderModalService;
        const customerName = document.getElementById('ord-name').value.trim();
        const pickup = document.getElementById('ord-pickup').value.trim();
        const dropoff = document.getElementById('ord-dropoff').value.trim();
        const paymentMethod = document.getElementById('ord-pay').value;
        const onlineDrv = state.drivers.find((d) => d.isOnline && d.approvalStatus === 'approved');
        const orderId = 'OZ-' + svc.slice(2, 3) + Math.floor(10000 + Math.random() * 89999);
        state.orders = [{
          orderId, serviceType: svc, customerName, pickupAddress: pickup, dropoffAddress: dropoff,
          summary: `${pickup} → ${dropoff}`, paymentMethod, status: onlineDrv ? 'assigned' : 'pending',
          assignedDriverId: onlineDrv ? onlineDrv.uid : null, assignedDriverName: onlineDrv ? onlineDrv.name : null,
          createdAtMs: Date.now(),
        }, ...state.orders];
        state.orderModalService = null;
        state.customerTab = 'pesanan';
        persistAll(); showToast(`Pesanan #${orderId} berhasil dibuat`, 'success'); render();
      } else if (e.target.id === 'order-chat-form') {
        const inp = document.getElementById('order-chat-input');
        sendOrderChat(state.activeChatOrderId, inp.value, state.activeChatRole);
      } else if (e.target.id === 'da-chat-form') {
        const inp = document.getElementById('da-chat-input');
        sendDriverAdminChat(state.activeDAChatDriverId, inp.value, state.activeDAChatRole);
      } else if (e.target.id === 'admin-edit-user-form') {
        const u = state.customers.find((c) => c.uid === state.editingCustomerUid);
        if (u) {
          u.name = document.getElementById('aeu-name').value.trim();
          u.phone = document.getElementById('aeu-phone').value.trim();
          u.normalizedPhone = normalizeWA(u.phone);
          u.address = document.getElementById('aeu-addr').value.trim();
          if (state.profile.uid === u.uid) state.profile = { uid: u.uid, name: u.name, phone: u.phone, address: u.address };
          state.editingCustomerUid = null;
          persistAll(); showToast(`Data user ${u.name} berhasil diubah`, 'success'); render();
        }
      } else if (e.target.id === 'admin-edit-driver-form') {
        const d = state.drivers.find((x) => x.uid === state.editingDriverUid);
        if (d) {
          d.name = document.getElementById('aed-name').value.trim();
          d.phone = document.getElementById('aed-phone').value.trim();
          d.vehicleModel = document.getElementById('aed-model').value.trim();
          d.area = document.getElementById('aed-area').value.trim();
          state.editingDriverUid = null;
          persistAll(); showToast(`Data driver ${d.name} berhasil diubah`, 'success'); render();
        }
      }
    });

    render();
